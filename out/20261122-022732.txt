[Unit]
Description=Internet for selected apps through a USB adapter, LTE stick, phone or Wi-Fi (mgu-uplink)
RequiresMountsFor=/var/opt
After=local-fs.target
# IterioDev's iPhone tether drives the same phone: never both at once
Conflicts=ipheth-tether.service

[Service]
ExecStart=/var/opt/mgu-uplink/current/mgu-uplink run -c /var/opt/mgu-uplink/current/mgu-uplink.conf
Restart=always
RestartSec=5
# The unit's system.conf sends stdout to null by default; mgu-uplink logs there.
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
# Written by mgu-deploy for every service a tool installs: a crash leaves no core dump for the
# crash handler, which would restart the head unit, a service that keeps failing stops, and
# it does not start in safe mode.
[Unit]
StartLimitIntervalSec=600
StartLimitBurst=20
ConditionPathExists=|!/var/opt/mgu-deploy/safe-mode
ConditionPathExists=|/run/mgu-deploy/safe-mode-lifted
ConditionPathExists=|!/var/opt/mgu-deploy/current/safe_mode.sh
After=mgu-deploy-safe-mode.service

[Service]
LimitCORE=1
